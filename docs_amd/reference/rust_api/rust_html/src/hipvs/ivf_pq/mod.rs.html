28//! Inverted File Product Quantization
29//!
30//! Example:
31//! ```
32//!
33//! use hipvs::ivf_pq::{Index, IndexParams, SearchParams};
34//! use hipvs::{ManagedTensor, Resources, Result};
35//!
36//! use ndarray::s;
37//! use ndarray_rand::rand_distr::Uniform;
38//! use ndarray_rand::RandomExt;
39//!
40//! fn ivf_pq_example() -> Result<()> {
41//!     let res = Resources::new()?;
42//!
43//!     // Create a new random dataset to index
44//!     let n_datapoints = 65536;
45//!     let n_features = 512;
46//!     let dataset =
47//!         ndarray::Array::<f32, _>::random((n_datapoints, n_features), Uniform::new(0., 1.0));
48//!
49//!     // build the ivf-pq index
50//!     let build_params = IndexParams::new()?;
51//!     let index = Index::build(&res, &build_params, &dataset)?;
52//!     println!(
53//!         "Indexed {}x{} datapoints into ivf-pq index",
54//!         n_datapoints, n_features
55//!     );
56//!
57//!     // use the first 4 points from the dataset as queries : will test that we get them back
58//!     // as their own nearest neighbor
59//!     let n_queries = 4;
60//!     let queries = dataset.slice(s![0..n_queries, ..]);
61//!
62//!     let k = 10;
63//!
64//!     // Ivf-Pq search API requires queries and outputs to be on device memory
65//!     // copy query data over, and allocate new device memory for the distances/ neighbors
66//!     // outputs
67//!     let queries = ManagedTensor::from(&queries).to_device(&res)?;
68//!     let mut neighbors_host = ndarray::Array::<u32, _>::zeros((n_queries, k));
69//!     let neighbors = ManagedTensor::from(&neighbors_host).to_device(&res)?;
70//!
71//!     let mut distances_host = ndarray::Array::<f32, _>::zeros((n_queries, k));
72//!     let distances = ManagedTensor::from(&distances_host).to_device(&res)?;
73//!
74//!     let search_params = SearchParams::new()?;
75//!
76//!     index.search(&res, &search_params, &queries, &neighbors, &distances)?;
77//!
78//!     // Copy back to host memory
79//!     distances.to_host(&res, &mut distances_host)?;
80//!     neighbors.to_host(&res, &mut neighbors_host)?;
81//!
82//!     // nearest neighbors should be themselves, since queries are from the
83//!     // dataset
84//!     println!("Neighbors {:?}", neighbors_host);
85//!     println!("Distances {:?}", distances_host);
86//!     Ok(())
87//! }
88//! ```
