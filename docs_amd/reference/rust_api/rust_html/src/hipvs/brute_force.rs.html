23//! Brute Force KNN
24
25use std::io::{stderr, Write};
26
27use crate::distance_type::DistanceType;
28use crate::dlpack::ManagedTensor;
29use crate::error::{check_cuvs, Result};
30use crate::resources::Resources;
31
32/// Brute Force KNN Index
33#[derive(Debug)]
34pub struct Index(ffi::cuvsBruteForceIndex_t);
35
36impl Index {
37    /// Builds a new Brute Force KNN Index from the dataset for efficient search.
38    ///
39    /// # Arguments
40    ///
41    /// * `res` - Resources to use
42    /// * `metric` - DistanceType to use for building the index
43    /// * `metric_arg` - Optional value of `p` for Minkowski distances
44    /// * `dataset` - A row-major matrix on either the host or device to index
45    pub fn build<T: Into<ManagedTensor>>(
46        res: &Resources,
47        metric: DistanceType,
48        metric_arg: Option<f32>,
49        dataset: T,
50    ) -> Result<Index> {
51        let dataset: ManagedTensor = dataset.into();
52        let index = Index::new()?;
53        unsafe {
54            check_cuvs(ffi::cuvsBruteForceBuild(
55                res.0,
56                dataset.as_ptr(),
57                metric,
58                metric_arg.unwrap_or(2.0),
59                index.0,
60            ))?;
61        }
62        Ok(index)
63    }
64
65    /// Creates a new empty index
66    pub fn new() -> Result<Index> {
67        unsafe {
68            let mut index = std::mem::MaybeUninit::<ffi::cuvsBruteForceIndex_t>::uninit();
69            check_cuvs(ffi::cuvsBruteForceIndexCreate(index.as_mut_ptr()))?;
70            Ok(Index(index.assume_init()))
71        }
72    }
73
74    /// Perform a Nearest Neighbors search on the Index
75    ///
76    /// # Arguments
77    ///
78    /// * `res` - Resources to use
79    /// * `queries` - A matrix in device memory to query for
80    /// * `neighbors` - Matrix in device memory that receives the indices of the nearest neighbors
81    /// * `distances` - Matrix in device memory that receives the distances of the nearest neighbors
82    pub fn search(
83        &self,
84        res: &Resources,
85        queries: &ManagedTensor,
86        neighbors: &ManagedTensor,
87        distances: &ManagedTensor,
88    ) -> Result<()> {
89        unsafe {
90            let prefilter = ffi::cuvsFilter {
91                addr: 0,
92                type_: ffi::cuvsFilterType::NO_FILTER,
93            };
94
95            check_cuvs(ffi::cuvsBruteForceSearch(
96                res.0,
97                self.0,
98                queries.as_ptr(),
99                neighbors.as_ptr(),
100                distances.as_ptr(),
101                prefilter,
102            ))
103        }
104    }
105}
106
107impl Drop for Index {
108    fn drop(&mut self) {
109        if let Err(e) = check_cuvs(unsafe { ffi::cuvsBruteForceIndexDestroy(self.0) }) {
110            write!(stderr(), "failed to call bruteForceIndexDestroy {:?}", e)
111                .expect("failed to write to stderr");
112        }
113    }
114}

116#[cfg(test)]
117mod tests {
118    use super::*;
119    use mark_flaky_tests::flaky;
120    use ndarray::s;
121    use ndarray_rand::rand_distr::Uniform;
122    use ndarray_rand::RandomExt;
123
124    fn test_bfknn(metric: DistanceType) {
125        let res = Resources::new().unwrap();
126
127        // Create a new random dataset to index
128        let n_datapoints = 16;
129        let n_features = 8;
130        let dataset_host =
131            ndarray::Array::<f32, _>::random((n_datapoints, n_features), Uniform::new(0., 1.0));
132
133        let dataset = ManagedTensor::from(&dataset_host).to_device(&res).unwrap();
134
135        println!("dataset {:#?}", dataset_host);
136
137        // build the brute force index
138        let index =
139            Index::build(&res, metric, None, dataset).expect("failed to create brute force index");
140
141        res.sync_stream().unwrap();
142
143        // use the first 4 points from the dataset as queries : will test that we get them back
144        // as their own nearest neighbor
145        let n_queries = 4;
146        let queries = dataset_host.slice(s![0..n_queries, ..]);
147
148        let k = 4;
149
150        println!("queries! {:#?}", queries);
151        let queries = ManagedTensor::from(&queries).to_device(&res).unwrap();
152        let mut neighbors_host = ndarray::Array::<i64, _>::zeros((n_queries, k));
153        let neighbors = ManagedTensor::from(&neighbors_host)
154            .to_device(&res)
155            .unwrap();
156
157        let mut distances_host = ndarray::Array::<f32, _>::zeros((n_queries, k));
158        let distances = ManagedTensor::from(&distances_host)
159            .to_device(&res)
160            .unwrap();
161
162        index
163            .search(&res, &queries, &neighbors, &distances)
164            .unwrap();
165
166        // Copy back to host memory
167        distances.to_host(&res, &mut distances_host).unwrap();
168        neighbors.to_host(&res, &mut neighbors_host).unwrap();
169        res.sync_stream().unwrap();
170
171        println!("distances {:#?}", distances_host);
172        println!("neighbors {:#?}", neighbors_host);
173
174        // nearest neighbors should be themselves, since queries are from the
175        // dataset
176        assert_eq!(neighbors_host[[0, 0]], 0);
177        assert_eq!(neighbors_host[[1, 0]], 1);
178        assert_eq!(neighbors_host[[2, 0]], 2);
179        assert_eq!(neighbors_host[[3, 0]], 3);
180    }
