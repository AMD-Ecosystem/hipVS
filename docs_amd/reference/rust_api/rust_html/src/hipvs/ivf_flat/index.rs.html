24use std::io::{stderr, Write};
25
26use crate::dlpack::ManagedTensor;
27use crate::error::{check_cuvs, Result};
28use crate::ivf_flat::{IndexParams, SearchParams};
29use crate::resources::Resources;
30
31/// Ivf-Flat ANN Index
32#[derive(Debug)]
33pub struct Index(ffi::cuvsIvfFlatIndex_t);
34
35impl Index {
36    /// Builds a new Index from the dataset for efficient search.
37    ///
38    /// # Arguments
39    ///
40    /// * `res` - Resources to use
41    /// * `params` - Parameters for building the index
42    /// * `dataset` - A row-major matrix on either the host or device to index
43    pub fn build<T: Into<ManagedTensor>>(
44        res: &Resources,
45        params: &IndexParams,
46        dataset: T,
47    ) -> Result<Index> {
48        let dataset: ManagedTensor = dataset.into();
49        let index = Index::new()?;
50        unsafe {
51            check_cuvs(ffi::cuvsIvfFlatBuild(
52                res.0,
53                params.0,
54                dataset.as_ptr(),
55                index.0,
56            ))?;
57        }
58        Ok(index)
59    }
60
61    /// Creates a new empty index
62    pub fn new() -> Result<Index> {
63        unsafe {
64            let mut index = std::mem::MaybeUninit::<ffi::cuvsIvfFlatIndex_t>::uninit();
65            check_cuvs(ffi::cuvsIvfFlatIndexCreate(index.as_mut_ptr()))?;
66            Ok(Index(index.assume_init()))
67        }
68    }
69
70    /// Perform a Approximate Nearest Neighbors search on the Index
71    ///
72    /// # Arguments
73    ///
74    /// * `res` - Resources to use
75    /// * `params` - Parameters to use in searching the index
76    /// * `queries` - A matrix in device memory to query for
77    /// * `neighbors` - Matrix in device memory that receives the indices of the nearest neighbors
78    /// * `distances` - Matrix in device memory that receives the distances of the nearest neighbors
79    pub fn search(
80        &self,
81        res: &Resources,
82        params: &SearchParams,
83        queries: &ManagedTensor,
84        neighbors: &ManagedTensor,
85        distances: &ManagedTensor,
86    ) -> Result<()> {
87        unsafe {
88            let prefilter = ffi::cuvsFilter {
89                addr: 0,
90                type_: ffi::cuvsFilterType::NO_FILTER,
91            };
92
93            check_cuvs(ffi::cuvsIvfFlatSearch(
94                res.0,
95                params.0,
96                self.0,
97                queries.as_ptr(),
98                neighbors.as_ptr(),
99                distances.as_ptr(),
100                prefilter,
101            ))
102        }
103    }
104}
105
106impl Drop for Index {
107    fn drop(&mut self) {
108        if let Err(e) = check_cuvs(unsafe { ffi::cuvsIvfFlatIndexDestroy(self.0) }) {
109            write!(stderr(), "failed to call cuvsIvfFlatIndexDestroy {:?}", e)
110                .expect("failed to write to stderr");
111        }
112    }
113}

115#[cfg(test)]
116mod tests {
117    use super::*;
118    use ndarray::s;
119    use ndarray_rand::rand_distr::Uniform;
120    use ndarray_rand::RandomExt;
121
122    #[test]
123    fn test_ivf_flat() {
124        let build_params = IndexParams::new().unwrap().set_n_lists(64);
125
126        let res = Resources::new().unwrap();
127
128        // Create a new random dataset to index
129        let n_datapoints = 1024;
130        let n_features = 16;
131        let dataset =
132            ndarray::Array::<f32, _>::random((n_datapoints, n_features), Uniform::new(0., 1.0));
133
134        let dataset_device = ManagedTensor::from(&dataset).to_device(&res).unwrap();
135
136        // build the ivf-flat index
137        let index = Index::build(&res, &build_params, dataset_device)
138            .expect("failed to create ivf-flat index");
139
140        // use the first 4 points from the dataset as queries : will test that we get them back
141        // as their own nearest neighbor
142        let n_queries = 4;
143        let queries = dataset.slice(s![0..n_queries, ..]);
144
145        let k = 10;
146
147        // IvfFlat search API requires queries and outputs to be on device memory
148        // copy query data over, and allocate new device memory for the distances/ neighbors
149        // outputs
150        let queries = ManagedTensor::from(&queries).to_device(&res).unwrap();
151        let mut neighbors_host = ndarray::Array::<i64, _>::zeros((n_queries, k));
152        let neighbors = ManagedTensor::from(&neighbors_host)
153            .to_device(&res)
154            .unwrap();
155
156        let mut distances_host = ndarray::Array::<f32, _>::zeros((n_queries, k));
157        let distances = ManagedTensor::from(&distances_host)
158            .to_device(&res)
159            .unwrap();
160
161        let search_params = SearchParams::new().unwrap();
162
163        index
164            .search(&res, &search_params, &queries, &neighbors, &distances)
165            .unwrap();
166
167        // Copy back to host memory
168        distances.to_host(&res, &mut distances_host).unwrap();
169        neighbors.to_host(&res, &mut neighbors_host).unwrap();
170
171        // nearest neighbors should be themselves, since queries are from the
172        // dataset
173        assert_eq!(neighbors_host[[0, 0]], 0);
174        assert_eq!(neighbors_host[[1, 0]], 1);
175        assert_eq!(neighbors_host[[2, 0]], 2);
176        assert_eq!(neighbors_host[[3, 0]], 3);
177    }
178
179    /// Test that an index can be searched multiple times without rebuilding.
180    /// This validates that search() takes &self instead of self.
181    #[test]
182    fn test_ivf_flat_multiple_searches() {
183        let build_params = IndexParams::new().unwrap().set_n_lists(64);
184        let res = Resources::new().unwrap();
185
186        // Create a random dataset
187        let n_datapoints = 1024;
188        let n_features = 16;
189        let dataset =
190            ndarray::Array::<f32, _>::random((n_datapoints, n_features), Uniform::new(0., 1.0));
191
192        let dataset_device = ManagedTensor::from(&dataset).to_device(&res).unwrap();
193
194        // Build the index once
195        let index = Index::build(&res, &build_params, dataset_device)
196            .expect("failed to create ivf-flat index");
197
198        let search_params = SearchParams::new().unwrap();
199        let k = 5;
200
201        // Perform multiple searches on the same index
202        for search_iter in 0..3 {
203            let n_queries = 4;
204            let queries = dataset.slice(s![0..n_queries, ..]);
205            let queries = ManagedTensor::from(&queries).to_device(&res).unwrap();
206
207            let mut neighbors_host = ndarray::Array::<i64, _>::zeros((n_queries, k));
208            let neighbors = ManagedTensor::from(&neighbors_host)
209                .to_device(&res)
210                .unwrap();
211
212            let mut distances_host = ndarray::Array::<f32, _>::zeros((n_queries, k));
213            let distances = ManagedTensor::from(&distances_host)
214                .to_device(&res)
215                .unwrap();
216
217            // This should work on every iteration because search() takes &self
218            index
219                .search(&res, &search_params, &queries, &neighbors, &distances)
220                .expect(&format!("search iteration {} failed", search_iter));
221
222            // Copy back to host memory
223            distances.to_host(&res, &mut distances_host).unwrap();
224            neighbors.to_host(&res, &mut neighbors_host).unwrap();
225
226            // Verify results are consistent
227            assert_eq!(
228                neighbors_host[[0, 0]],
229                0,
230                "iteration {}: first query should find itself",
231                search_iter
232            );
233        }
234    }
