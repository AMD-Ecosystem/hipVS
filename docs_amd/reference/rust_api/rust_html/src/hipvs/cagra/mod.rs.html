28//! CAGRA is a graph-based nearest neighbors implementation with state-of-the art
29//! query performance for both small- and large-batch sized search.
30//!
31//! Example:
32//! ```
33//!
34//! use hipvs::cagra::{Index, IndexParams, SearchParams};
35//! use hipvs::{ManagedTensor, Resources, Result};
36//!
37//! use ndarray::s;
38//! use ndarray_rand::rand_distr::Uniform;
39//! use ndarray_rand::RandomExt;
40//!
41//! fn cagra_example() -> Result<()> {
42//!     let res = Resources::new()?;
43//!
44//!     // Create a new random dataset to index
45//!     let n_datapoints = 65536;
46//!     let n_features = 512;
47//!     let dataset =
48//!         ndarray::Array::<f32, _>::random((n_datapoints, n_features), Uniform::new(0., 1.0));
49//!
50//!     // build the cagra index
51//!     let build_params = IndexParams::new()?;
52//!     let index = Index::build(&res, &build_params, &dataset)?;
53//!     println!(
54//!         "Indexed {}x{} datapoints into cagra index",
55//!         n_datapoints, n_features
56//!     );
57//!
58//!     // use the first 4 points from the dataset as queries : will test that we get them back
59//!     // as their own nearest neighbor
60//!     let n_queries = 4;
61//!     let queries = dataset.slice(s![0..n_queries, ..]);
62//!
63//!     let k = 10;
64//!
65//!     // CAGRA search API requires queries and outputs to be on device memory
66//!     // copy query data over, and allocate new device memory for the distances/ neighbors
67//!     // outputs
68//!     let queries = ManagedTensor::from(&queries).to_device(&res)?;
69//!     let mut neighbors_host = ndarray::Array::<u32, _>::zeros((n_queries, k));
70//!     let neighbors = ManagedTensor::from(&neighbors_host).to_device(&res)?;
71//!
72//!     let mut distances_host = ndarray::Array::<f32, _>::zeros((n_queries, k));
73//!     let distances = ManagedTensor::from(&distances_host).to_device(&res)?;
74//!
75//!     let search_params = SearchParams::new()?;
76//!
77//!     index.search(&res, &search_params, &queries, &neighbors, &distances)?;
78//!
79//!     // Copy back to host memory
80//!     distances.to_host(&res, &mut distances_host)?;
81//!     neighbors.to_host(&res, &mut neighbors_host)?;
82//!
83//!     // nearest neighbors should be themselves, since queries are from the
84//!     // dataset
85//!     println!("Neighbors {:?}", neighbors_host);
86//!     println!("Distances {:?}", distances_host);
87//!     Ok(())
88//! }
89//! ```
90//!
91//! Serialization example:
92//! ```no_run
93//! use hipvs::cagra::{Index, IndexParams};
94//! use hipvs::{Resources, Result};
95//!
96//! fn serialize_example() -> Result<()> {
97//!     let res = Resources::new()?;
98//!
99//!     // Build an index (using some dataset)
100//!     let build_params = IndexParams::new()?;
101//!     // let index = Index::build(&res, &build_params, &dataset)?;
102//!
103//!     // Save the index to disk (including the dataset)
104//!     // index.serialize(&res, "/path/to/index.bin", true)?;
105//!
106//!     // Later, load the index from disk
107//!     let loaded_index = Index::deserialize(&res, "/path/to/index.bin")?;
108//!
109//!     // The loaded index can be used for search just like the original
110//!     Ok(())
111//! }
112//! ```
