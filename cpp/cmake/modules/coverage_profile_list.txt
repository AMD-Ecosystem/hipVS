# Passed to -fprofile-list by ENABLE_CODE_COVERAGE.
#
# Device profiling stores each instrumented function's address in its __llvm_prf_data record, so
# every instrumented device function counts as address-taken. AMDGPU's LDS lowering then allocates
# the __shared__ variables of all address-taken functions in every kernel that makes an unknown
# call (inline asm counts), which pushes kernels past the 64 KiB LDS limit. Keep non-kernel device
# functions that use __shared__ directly out of the instrumentation. Use the forbid form (!fun:);
# "=skip" still emits the data record.

default:allow
